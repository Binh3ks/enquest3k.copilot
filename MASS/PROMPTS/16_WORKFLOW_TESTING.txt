## LEVEL 2: INTEGRATION TESTING (5 minutes)

### Test 2.1: Vocab → Read Cross-Reference
```bash
node -e "
const v = require('./src/data/weeks/week_XX/vocab.js').default;
const r = require('./src/data/weeks/week_XX/read.js').default;

const vocabWords = v.vocab.map(w => w.word);
const boldedWords = (r.content_en.match(/\*\*(.*?)\*\*/g) || []).map(w => w.replace(/\*\*/g, ''));

console.log('Vocab words:', vocabWords.join(', '));
console.log('Bolded in read:', boldedWords.join(', '));

const coverage = vocabWords.filter(w => boldedWords.includes(w));
console.log('Coverage:', coverage.length, '/', vocabWords.length, '(' + Math.round(coverage.length/vocabWords.length*100) + '%)');
console.log('Rule: At least 50%:', coverage.length >= vocabWords.length * 0.5 ? '✓' : '✗');
"
```

### Test 2.2: Read → Dictation Sentence Match
```bash
node -e "
const r = require('./src/data/weeks/week_XX/read.js').default;
const d = require('./src/data/weeks/week_XX/dictation.js').default;

// Extract sentences from read (split by '. ')
const readSentences = r.content_en.replace(/\*\*/g, '').split('. ').map(s => s.trim()).filter(s => s.length > 0);
const dictationSentences = d.sentences.map(s => s.text);

console.log('Read sentences:', readSentences.length);
console.log('Dictation sentences:', dictationSentences.length);
console.log('Match count:', readSentences.length === dictationSentences.length ? '✓' : '✗ MISMATCH');

// Sample first sentence
console.log('\nSample comparison:');
console.log('Read:', readSentences[0]);
console.log('Dictation:', dictationSentences[0]);
console.log('Match:', readSentences[0] === dictationSentences[0] ? '✓' : '✗');
"
```

### Test 2.3: Dictation → Shadowing Match
```bash
node -e "
const d = require('./src/data/weeks/week_XX/dictation.js').default;
const s = require('./src/data/weeks/week_XX/shadowing.js').default;

console.log('Dictation count:', d.sentences.length);
console.log('Shadowing count:', s.sentences.length);
console.log('Count match:', d.sentences.length === s.sentences.length ? '✓' : '✗');

// Check first 3 sentences
for (let i = 0; i < Math.min(3, d.sentences.length); i++) {

}

// Check shadowing has audio_url
console.log('Shadowing has audio:', s.sentences[0].audio_url ? '✓' : '✗');
"
```

### Test 2.4: Vocab → Word Match Complete Set
```bash
node -e "
const v = require('./src/data/weeks/week_XX/vocab.js').default;
const wm = require('./src/data/weeks/week_XX/word_match.js').default;

const vocabWords = v.vocab.map(w => w.word).sort();

```
