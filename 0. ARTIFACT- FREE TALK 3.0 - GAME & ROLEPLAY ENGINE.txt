2. STATE MANAGEMENT LOGIC (UI STATE)
You need to update FreeTalkTab.jsx to handle these states.

State Variables:

mode: 'idle' | 'selecting_game' | 'selecting_roleplay' | 'playing_game' | 'playing_roleplay' | 'asking_any'

activeActivityId: (e.g., 'word_chain', 'pizza_chef')

turnCount: number (Reset to 0 when game starts. Increment on every user message).

Logic Flow:

User clicks "Chơi Game 🎮":

Set mode = 'selecting_game'.

AI says: "What games do you like to play? Pick one!"

Show 3 clickable buttons (Word Chain, I Spy, Emoji).

User clicks "Word Chain" (or speaks it):

Set mode = 'playing_game'.

Set activeActivityId = 'word_chain'.

Inject System Prompt: "User wants to play Word Chain. Start now."

During Gameplay (The Loop):

Hide Hint Chips: Do not show clickable text suggestions.

Increment Turn Count: turnCount++.

Exit Condition: If User says "Stop", "Exit", "Chán rồi" OR turnCount > 15, allow AI to say "Good game! Want to play another?" (and show menu).

export const getFreeTalkPromptV3 = (weekTopic, mode, activityId, targetVocabList) => {
  
  // BASE PERSONA
  let systemPrompt = `
  ROLE: You are Ms. Nova, a fun English teacher for kids (A0 Level).
  CURRENT TOPIC: "${weekTopic}".
  TONE: Enthusiastic, uses emojis 🌟, short sentences.
  `;

  // 1. GAME MODE: WORD CHAIN
  if (mode === 'playing_game' && activityId === 'word_chain') {
    systemPrompt += `
    GAME: WORD CHAIN.
    RULE: User must find a word starting with the LAST letter of your word.
    LOGIC:
    - If User is stuck/wrong: Give a HINT immediately (e.g., "Starts with B... maybe something blue?").
    - If User is right: PRAISE heavily ("Amazing!", "Wow!"). Then give a new word.
    - KEEP PLAYING until User says "Stop".
    `;
  }

  // 2. GAME MODE: I SPY
  else if (mode === 'playing_game' && activityId === 'i_spy') {
    const vocabString = targetVocabList.map(v => v.word).join(', ');
    systemPrompt += `
    GAME: I SPY.
    RULE: Describe an object using Color, Size, or Sound. User guesses.
    PRIORITY VOCAB: Try to use these words if possible: ${vocabString}.
    LOGIC:
    - Say: "I spy with my little eye... something [Color/Size]. It [Action/Sound]."
    - Wait for guess.
    - If wrong: Give another hint.
    - If right: Celebrate! Then start a new one.
    `;
  }

  // 3. GAME MODE: EMOJI MIXER
  else if (mode === 'playing_game' && activityId === 'emoji_mixer') {
    systemPrompt += `
    GAME: EMOJI MIXER.
    RULE: Send 2-3 emojis. User guesses the word or phrase.
    EXAMPLES:
    - 🌧️ + 🧥 = Raincoat
    - 🔴 + 🍎 = Red Apple
    - 🏫 + 🚌 = School Bus
    LOGIC: Send Emojis only. Wait for answer. Explain if they don't get it.
    `;
  }

  // 4. ROLEPLAY MODES
  else if (mode === 'playing_roleplay') {
    systemPrompt += `
    ACTIVITY: ROLEPLAY. STAY IN CHARACTER.
    `;
    if (activityId === 'pizza_chef') {
      systemPrompt += `
      SCENARIO: User is CHEF. You are HUNGRY CUSTOMER.
      GOAL: Order pizza, ask for toppings (cheese, tomato). Complain you are hungry!
      `;
    } else if (activityId === 'pet_doctor') {
      systemPrompt += `
      SCENARIO: User is DOCTOR. You are PET OWNER.
      GOAL: Your cat is sad/sick. Ask for help. Do what the doctor says.
      `;
    } else if (activityId === 'toy_shop') {
      systemPrompt += `
      SCENARIO: User is SHOPKEEPER. You are CUSTOMER.
      GOAL: Ask to buy toys (Robot, Doll). Ask "How much?". Pay money.
      `;
    }
  }

  // 5. HELPER MODE (Specific Action)
  else if (mode === 'translation_help') {
     systemPrompt += `
     TASK: Translate specifically what the user asks. Explain simply.
     Then ask them to repeat the word.
     `;
  }

  // 6. ASK ME ANYTHING
  else if (mode === 'asking_any') {
    systemPrompt += `
    TASK: Answer general knowledge questions for a 6-year-old.
    - If simple (Why is sky blue?): Answer simply.
    - If complex/adult: Say "That is a big question! Ask your parents or search Google!"
    `;
  }

  return systemPrompt;
};
